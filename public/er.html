    USER {
        string id PK
        string name "昵称"
        string avatar "emoji头像"
        string building "楼栋"
        string bio "简介"
        datetime createdAt
    }
    ITEM {
        string id PK
        string title "名称"
        string description "描述"
        string tradeMode "FREE|ANY|PRICED"
        float price "标价"
        string condition "成色"
        string images "图片JSON"
        string status "ACTIVE|GIVEN"
        string givenToId FK "送给谁"
        datetime givenAt "送出时间"
        string publisherId FK "发布者"
        datetime createdAt "新鲜度依据"
    }
    WANT {
        string id PK
        string itemId FK
        string userId FK
        datetime createdAt
    }
    COMMENT {
        string id PK
        string itemId FK
        string userId FK
        string content "内容"
        datetime createdAt
    }
